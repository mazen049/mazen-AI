// ========================================== // MAZEN AI - FINAL CLEAN MASTER SCRIPT // ========================================== document.addEventListener("DOMContentLoaded", () => { console.log("Mazen AI Engine Loaded Successfully."); // تفعيل التفاعل الأول تلقائياً لتجاوز قيود متصفحات الحماية على الصوت document.body.addEventListener('click', () => { if ('speechSynthesis' in window) { window.speechSynthesis.resume(); } }, { once: true }); setupMazenInterface(); }); // 1. نظام القراءة الصوتية (Text-to-Speech) function speakResponse(text) { if ('speechSynthesis' in window) { window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'ar-SA'; utterance.rate = 1.0; utterance.pitch = 1.0; window.speechSynthesis.speak(utterance); } else { console.warn("Speech synthesis is not supported in this browser."); } } // 2. نظام التنزيل الحقيقي للملفات (Download Engine) function downloadFile(content, filename, contentType) { try { const blob = new Blob([content], { type: contentType }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = filename; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); console.log("File downloaded successfully:", filename); } catch (error) { console.error("Download failed:", error); alert("تعذر تنزيل الملف، تحقق من صلاحيات المتصفح."); } } // 3. نظام توليد الصور والرسومات (Image Generation Engine) async function generateAIImage(promptText) { let imageContainer = document.getElementById('image-output-container') || document.getElementById('chat-container'); if (!imageContainer) { imageContainer = document.body; } const loadingMsg = document.createElement('div'); loadingMsg.className = 'ai-loading-message'; loadingMsg.style.color = '#00ffff'; loadingMsg.style.margin = '10px 0'; loadingMsg.innerText = 'جاري توليد الصورة بالذكاء الاصطناعي، يرجى الانتظار...'; imageContainer.appendChild(loadingMsg); try { setTimeout(() => { loadingMsg.remove(); const img = document.createElement('img'); img.src = `https://image.pollinations.ai/prompt/${encodeURIComponent(promptText)}`; img.alt = promptText; img.style.maxWidth = '100%'; img.style.borderRadius = '12px'; img.style.marginTop = '10px'; img.style.boxShadow = '0 0 15px rgba(0, 255, 255, 0.3)'; imageContainer.appendChild(img); speakResponse("ها هي الصورة وجدات بنجاح!"); }, 2500); } catch (error) { loadingMsg.remove(); console.error("Image generation error:", error); speakResponse("عذراً، حدث خطأ أثناء توليد الصورة."); } } // ربط الأزرار والعناصر تلقائياً بالواجهة function setupMazenInterface() { const downloadBtn = document.getElementById('download-btn'); if (downloadBtn) { downloadBtn.addEventListener('click', () => { const textToSave = document.getElementById('output-text')?.innerText || "محتوى من منصة MAZEN AI"; downloadFile(textToSave, 'MazenAI-Output.txt', 'text/plain;charset=utf-8'); }); } const imageBtn = document.getElementById('generate-image-btn'); if (imageBtn) { imageBtn.addEventListener('click', () => { const promptInput = document.getElementById('user-input')?.value || "A futuristic AI cyberpunk city"; generateAIImage(promptInput); }); } } // تصدير الوظائف للاستخدام العام في أزرار HTML window.MazenAI = { speak: speakResponse, download: downloadFile, generateImage: generateAIImage };
